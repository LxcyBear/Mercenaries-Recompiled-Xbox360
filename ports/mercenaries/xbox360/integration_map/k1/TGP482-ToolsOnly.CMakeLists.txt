# TGP482 XboxXBERecomp tools only: no Windows runtime, SDL, ImGui, Direct3D12.
# Original source under SDK_DIR is used unchanged; generated guest data
# belongs in a separate private user-local workspace, never the game repo.
cmake_minimum_required(VERSION 3.25)
project(Merc360_TGP482_AnalysisTools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT SDK_DIR OR NOT EXISTS "${SDK_DIR}/tools/recomp/main.cpp")
  message(FATAL_ERROR "SDK_DIR must point to a local TGP482/XboxXBERecomp checkout")
endif()
include(FetchContent)
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zydis GIT_REPOSITORY https://github.com/zyantific/zydis.git GIT_TAG v4.1.1 GIT_SHALLOW TRUE)
FetchContent_Declare(tomlplusplus GIT_REPOSITORY https://github.com/marzer/tomlplusplus.git GIT_TAG v3.4.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(zydis tomlplusplus)
file(GLOB XBE_RECOMP_SOURCES CONFIGURE_DEPENDS "${SDK_DIR}/tools/recomp/*.cpp")
add_executable(xbe_recomp ${XBE_RECOMP_SOURCES})
target_include_directories(xbe_recomp PRIVATE "${SDK_DIR}/runtime" "${SDK_DIR}/tools/recomp")
target_compile_definitions(xbe_recomp PRIVATE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(xbe_recomp PRIVATE Zydis tomlplusplus::tomlplusplus)
add_executable(xbe_symbols "${SDK_DIR}/tools/symbols/main.cpp")
target_include_directories(xbe_symbols PRIVATE "${SDK_DIR}/runtime")
target_compile_definitions(xbe_symbols PRIVATE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(xbe_symbols PRIVATE Zydis tomlplusplus::tomlplusplus)
